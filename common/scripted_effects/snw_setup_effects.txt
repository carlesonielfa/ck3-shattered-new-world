# Shattered New World - top-level setup
# Runs once from on_game_start_after_lobby. Order matters:
#   1. remove titles that only make sense for historical people (noble families, adventurer camps)
#   2. destroy titles above the chosen tier and make the now-invalid vassals independent
#   3. (Every County) split multi-county rulers
#   4. replace every remaining historical ruler with a generated one, keeping the realm structure
#   5. give the new rulers families, courts, resources and game-start setup
#   6. remove every remaining historical character

snw_setup_effect = {
	snw_remove_historical_only_titles_effect = yes
	snw_dissolve_high_titles_effect = yes
	snw_fix_vassalage_effect = yes
	snw_fix_vassalage_effect = yes	# second pass catches vassals whose liege changed in the first pass
	if = {
		limit = { has_game_rule = snw_shatter_counties }
		snw_split_counties_effect = yes
	}
	snw_replace_all_rulers_effect = yes
	every_in_list = {
		list = snw_new_rulers
		snw_finalize_ruler_effect = yes
	}
	snw_hand_over_players_effect = yes
	snw_remove_historical_characters_effect = yes
	set_global_variable = { name = snw_world_generated value = yes }
}

### 1. Titles tied to historical people

snw_remove_historical_only_titles_effect = {
	every_living_character = {
		limit = {
			any_held_title = {
				OR = {
					is_noble_family_title = yes
					AND = {
						is_landless_type_title = yes
						holder = { government_has_flag = government_is_landless_adventurer }
					}
				}
			}
		}
		save_temporary_scope_as = snw_title_owner
		every_held_title = {
			limit = {
				OR = {
					is_noble_family_title = yes
					AND = {
						is_landless_type_title = yes
						holder = { government_has_flag = government_is_landless_adventurer }
					}
				}
			}
			save_temporary_scope_as = snw_doomed_title
			scope:snw_title_owner = { destroy_title = scope:snw_doomed_title }
		}
	}
}

### 2. Breaking up realms

snw_destroy_this_title_effect = {
	save_temporary_scope_as = snw_doomed_title
	holder = { destroy_title = scope:snw_doomed_title }
}

snw_dissolve_high_titles_effect = {
	# Every shatter level removes empires
	every_empire = {
		limit = { snw_dissolvable_title_trigger = yes }
		snw_destroy_this_title_effect = yes
	}
	if = {
		limit = { NOT = { has_game_rule = snw_shatter_empires } }
		every_kingdom = {
			limit = { snw_dissolvable_title_trigger = yes }
			snw_destroy_this_title_effect = yes
		}
	}
	if = {
		limit = {
			OR = {
				has_game_rule = snw_shatter_duchies
				has_game_rule = snw_shatter_counties
			}
		}
		every_duchy = {
			limit = { snw_dissolvable_title_trigger = yes }
			snw_destroy_this_title_effect = yes
		}
	}
}

snw_fix_vassalage_effect = {
	every_ruler = {
		limit = { snw_must_become_independent_trigger = yes }
		add_to_temporary_list = snw_going_independent
	}
	every_in_list = {
		list = snw_going_independent
		limit = { snw_must_become_independent_trigger = yes }
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = snw_change
			add_claim_on_loss = no
		}
		becomes_independent = { change = scope:snw_change }
		resolve_title_and_vassal_change = scope:snw_change
	}
	# Barons whose liege is no longer above them go to the holder of their county
	every_ruler = {
		limit = {
			highest_held_title_tier = tier_barony
			is_independent_ruler = no
			exists = liege
			liege.highest_held_title_tier <= tier_barony
			exists = capital_county.holder
			NOT = { capital_county.holder = this }
			capital_county.holder.highest_held_title_tier >= tier_county
		}
		capital_county.holder = { save_temporary_scope_as = snw_new_liege }
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = snw_change
			add_claim_on_loss = no
		}
		change_liege = {
			liege = scope:snw_new_liege
			change = scope:snw_change
		}
		resolve_title_and_vassal_change = scope:snw_change
	}
}

### 3. Every County: one county per ruler

snw_split_counties_effect = {
	every_ruler = {
		limit = {
			snw_special_ruler_trigger = no
			any_held_title = {
				count >= 2
				tier = tier_county
			}
		}
		add_to_temporary_list = snw_multi_county_rulers
	}
	every_in_list = {
		list = snw_multi_county_rulers
		save_scope_as = snw_splitter
		snw_record_government_effect = yes
		snw_record_authority_laws_effect = yes
		every_held_title = {
			limit = {
				tier = tier_county
				NOT = { this = scope:snw_splitter.capital_county }
			}
			save_scope_as = snw_split_county
			scope:snw_splitter = {
				snw_spawn_ruler_effect = {
					SOURCE = scope:snw_splitter
					LAND = scope:snw_split_county
				}
			}
			create_title_and_vassal_change = {
				type = usurped
				save_scope_as = snw_change
				add_claim_on_loss = no
			}
			scope:snw_split_county = {
				change_title_holder_include_vassals = {
					holder = scope:snw_new
					change = scope:snw_change
					take_baronies = yes
				}
			}
			resolve_title_and_vassal_change = scope:snw_change
			scope:snw_new = {
				snw_apply_government_effect = { SOURCE = scope:snw_splitter }
				snw_apply_authority_laws_effect = { SOURCE = scope:snw_splitter }
			}
		}
	}
}

### 4. Replacing historical rulers

snw_replace_all_rulers_effect = {
	# every_living_character rather than every_ruler so baron-tier holders are caught too
	every_living_character = {
		limit = {
			NOT = { has_character_flag = snw_generated }
			any_held_title = { always = yes }
		}
		add_to_list = snw_old_rulers
	}
	every_in_list = {
		list = snw_old_rulers
		limit = {
			is_alive = yes
			any_held_title = { always = yes }
		}
		snw_replace_ruler_effect = yes
	}
}

# Scope: the historical ruler being replaced.
snw_replace_ruler_effect = {
	save_scope_as = snw_old
	clear_saved_scope = snw_old_liege
	if = {
		limit = {
			is_independent_ruler = no
			exists = liege
		}
		liege = { save_scope_as = snw_old_liege }
	}
	snw_record_government_effect = yes
	snw_record_authority_laws_effect = yes

	if = {
		limit = { exists = capital_county }
		snw_spawn_ruler_effect = {
			SOURCE = scope:snw_old
			LAND = scope:snw_old.capital_county
		}
	}
	else = {
		snw_spawn_ruler_effect = {
			SOURCE = scope:snw_old
			LAND = scope:snw_old.primary_title
		}
	}

	# Titles
	create_title_and_vassal_change = {
		type = usurped
		save_scope_as = snw_change
		add_claim_on_loss = no
	}
	every_held_title = {
		# Capital baronies move with their county
		limit = {
			NOT = {
				AND = {
					tier = tier_barony
					is_capital_barony = yes
				}
			}
		}
		change_title_holder = {
			holder = scope:snw_new
			change = scope:snw_change
		}
	}
	resolve_title_and_vassal_change = scope:snw_change

	# Vassals and liege: the new ruler takes the old one's place in the hierarchy
	create_title_and_vassal_change = {
		type = swear_fealty
		save_scope_as = snw_liege_change
		add_claim_on_loss = no
	}
	every_vassal = {
		change_liege = {
			liege = scope:snw_new
			change = scope:snw_liege_change
		}
	}
	if = {
		limit = {
			exists = scope:snw_old_liege
			scope:snw_old_liege = { is_alive = yes }
			NOT = { scope:snw_old_liege = scope:snw_new }
		}
		scope:snw_new = {
			change_liege = {
				liege = scope:snw_old_liege
				change = scope:snw_liege_change
			}
		}
	}
	resolve_title_and_vassal_change = scope:snw_liege_change

	scope:snw_new = {
		snw_apply_government_effect = { SOURCE = scope:snw_old }
		snw_apply_authority_laws_effect = { SOURCE = scope:snw_old }
	}

	# Keep regalia and court artifacts with the realm
	every_character_artifact = {
		set_owner = {
			target = scope:snw_new
			history = {
				type = inherited
				actor = scope:snw_old
				recipient = scope:snw_new
			}
		}
	}

	# Remember a player's replacement. The handover happens after families are generated
	# (see snw_hand_over_players_effect), so the deaths of the new rulers' parents are not
	# reported to the player.
	if = {
		limit = { is_ai = no }
		set_variable = {
			name = snw_replacement
			value = scope:snw_new
		}
	}
}

# Moves every player to the generated ruler who replaced their historical character.
snw_hand_over_players_effect = {
	every_in_list = {
		list = snw_old_rulers
		limit = {
			is_ai = no
			has_variable = snw_replacement
		}
		set_player_character = var:snw_replacement
	}
}
