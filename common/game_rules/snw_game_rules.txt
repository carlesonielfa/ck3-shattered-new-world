# How far historical realms are broken up.
snw_shatter_level = {
	categories = { game_modes }
	default = snw_shatter_kingdoms

	snw_shatter_empires = {}	# Empires destroyed; kingdoms are the largest independent realms
	snw_shatter_kingdoms = {}	# Empires and kingdoms destroyed; duchies are the largest realms
	snw_shatter_duchies = {}	# Everything above county destroyed; counts may still hold several counties
	snw_shatter_counties = {}	# Everything above county destroyed; every county gets its own ruler
}

	snw_ruler_age_young = {}	# 16-35
	snw_ruler_age_mature = {}	# 35-70
	snw_ruler_age_any = {}		# 2-75, child rulers possible
}
